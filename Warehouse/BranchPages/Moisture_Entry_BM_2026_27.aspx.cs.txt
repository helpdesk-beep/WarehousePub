    protected void ddlStack_SelectedIndexChanged(object sender, EventArgs e)
    {
        if (ddlStack.SelectedIndex <= 0)
            return;

        DataTable dt = ViewState["StackData"] as DataTable;

        if (dt == null)
            return;

        DataRow[] rows = dt.Select(
            "Stack_ID='" + ddlStack.SelectedValue.Replace("'", "''") + "'");

        if (rows.Length > 0)
        {
            lblStackId.Text = rows[0]["Stack_ID"].ToString();

            lblStackName.Text = rows[0]["Stack_Name"].ToString();

            lblBags.Text =
                (Convert.ToDecimal(rows[0]["recbags"])
                - Convert.ToDecimal(rows[0]["delbags"]))
                .ToString();

            lblQty.Text =
                (Convert.ToDecimal(rows[0]["recweght"])
                - Convert.ToDecimal(rows[0]["delwght"]))
                .ToString("0.00");

            txtStackCompletionDate.Text =
                rows[0]["Stack_Completion_Date"].ToString();

            lblCommodity.Text =
                rows[0]["Commodity_Name"].ToString();

            ViewState["CommodityId"] =
                rows[0]["Commodity_Id"].ToString();
        }
    }
    protected void btnSave_Click(object sender, EventArgs e)
    {
        try
        {
            DataTable dt = Session["TempStack"] as DataTable;

            if (dt == null || dt.Rows.Count == 0)
            {
                ShowMessage("No data to save");
                return;
            }

            SqlConnection con = new SqlConnection(conStr);
            con.Open();

            for (int i = 0; i < dt.Rows.Count; i++)
            {
                DataRow dr = dt.Rows[i];

                SqlCommand cmd = new SqlCommand(@"
                INSERT INTO tbl_FCI_Moisture_Entry_web
                (
                    BranchId, GodownId, CommodityId, StackId, StackName,
                    Bags, StackQty, MoistureDate, AvgMoisture,
                    SentFCI, SentDate, IsFCIChecked,
                    FCICheckDate, FCIMoisture,
                    BranchDocument, FCIDocument,
                    StackCompletionDate
                )
                VALUES
                (
                    @BranchId, @GodownId, @CommodityId, @StackId, @StackName,
                    @Bags, @StackQty, @MoistureDate, @AvgMoisture,
                    @SentFCI, @SentDate, @IsFCIChecked,
                    @FCICheckDate, @FCIMoisture,
                    @BranchDocument, @FCIDocument,
                    @StackCompletionDate
                )", con);

                cmd.Parameters.AddWithValue("@BranchId", branchid);
                cmd.Parameters.AddWithValue("@GodownId", dr["GodownId"]);
                cmd.Parameters.AddWithValue("@CommodityId", dr["CommodityId"]);
                cmd.Parameters.AddWithValue("@StackId", dr["StackId"]);
                cmd.Parameters.AddWithValue("@StackName", dr["StackName"]);

                cmd.Parameters.AddWithValue("@Bags", dr["Bags"]);
                cmd.Parameters.AddWithValue("@StackQty", dr["Qty"]);

                cmd.Parameters.AddWithValue("@MoistureDate", Convert.ToDateTime(dr["MoistureDate"]));
                cmd.Parameters.AddWithValue("@AvgMoisture", Convert.ToDecimal(dr["AvgMoisture"]));

                cmd.Parameters.AddWithValue("@SentFCI", dr["SentFCI"]);
                cmd.Parameters.AddWithValue("@SentDate", Convert.ToDateTime(dr["SentDate"]));
                cmd.Parameters.AddWithValue("@IsFCIChecked", dr["FCIChecked"]);

                if (dr["FCIChecked"].ToString() == "Y")
                {
                    cmd.Parameters.AddWithValue("@FCICheckDate", Convert.ToDateTime(dr["FCICheckDate"]));
                    cmd.Parameters.AddWithValue("@FCIMoisture", Convert.ToDecimal(dr["FCIMoisture"]));
                }
                else
                {
                    cmd.Parameters.AddWithValue("@FCICheckDate", DBNull.Value);
                    cmd.Parameters.AddWithValue("@FCIMoisture", DBNull.Value);
                }

                // Documents
                SqlParameter p1 = new SqlParameter("@BranchDocument", SqlDbType.VarBinary);
                if (dr["BranchDocument"] != DBNull.Value)
                    p1.Value = dr["BranchDocument"];
                else
                    p1.Value = DBNull.Value;

                cmd.Parameters.Add(p1);

                SqlParameter p2 = new SqlParameter("@FCIDocument", SqlDbType.VarBinary);
                if (dr["FCIDocument"] != DBNull.Value)
                    p2.Value = dr["FCIDocument"];
                else
                    p2.Value = DBNull.Value;

                cmd.Parameters.Add(p2);

                cmd.Parameters.AddWithValue("@StackCompletionDate", txtStackCompletionDate.Text);

                cmd.ExecuteNonQuery();
            }

            con.Close();

            Session["TempStack"] = null;
            gvTempStack.DataSource = null;
            gvTempStack.DataBind();

            LoadGrid();

            ShowMessage("Saved successfully");
        }
        catch (Exception ex)
        {
            ShowMessage(ex.Message);
        }
    }

    private bool IsPdfFile(FileUpload fu)
    {
        if (!fu.HasFile)
            return false;

        string ext = System.IO.Path.GetExtension(fu.FileName).ToLower();

        return ext == ".pdf";
    }
    private bool IsAlreadySaved()
    {
        bool result = false;

        using (SqlConnection con = new SqlConnection(conStr))
        {
            SqlCommand cmd = new SqlCommand(
            @"SELECT COUNT(*)
          FROM tbl_FCI_Moisture_Entry_web
          WHERE GodownId=@GodownId
          AND StackId=@StackId", con);

            cmd.Parameters.AddWithValue("@GodownId", ddlGodown.SelectedValue);
            cmd.Parameters.AddWithValue("@StackId", ddlStack.SelectedValue);

            con.Open();

            object obj = cmd.ExecuteScalar();

            int cnt = 0;

            if (obj != null)
            {
                cnt = Convert.ToInt32(obj);
            }

            if (cnt > 0)
            {
                result = true;
            }
        }

        return result;
    }

    protected void btnAddStack_Click(object sender, EventArgs e)
    {
        try
        {
            if (ddlGodown.SelectedIndex <= 0)
            {
                ShowMessage("Select Godown");
                return;
            }

            if (ddlStack.SelectedIndex <= 0)
            {
                ShowMessage("Select Stack");
                return;
            }

            if (txtMoistureDate.Text.Trim() == "")
            {
                ShowMessage("Select Moisture Date");
                return;
            }

            if (txtAvgMoisture.Text.Trim() == "")
            {
                ShowMessage("Enter Avg Moisture");
                return;
            }
            if (fuBranchDoc.HasFile)
            {
                if (!IsPdfFile(fuBranchDoc))
                {
                    ShowMessage("Branch document PDF only allowed");
                    return;
                }
            }

            if (fuFCIDoc.HasFile)
            {
                if (!IsPdfFile(fuFCIDoc))
                {
                    ShowMessage("FCI document PDF only allowed");
                    return;
                }
            }
            DataTable dt = GetTempTable();

            DataRow[] exist = dt.Select(
                "StackId='" + ddlStack.SelectedValue.Replace("'", "''") + "'");

            if (exist.Length > 0)
            {
                ShowMessage("Stack already added");
                return;
            }
            if (IsAlreadySaved())
            {
                ShowMessage("This stack already exists in database.");
                return;
            }
            byte[] branchDoc = null;
            byte[] fciDoc = null;

            if (fuBranchDoc.HasFile)
            {
                branchDoc = fuBranchDoc.FileBytes;
            }

            if (fuFCIDoc.HasFile)
            {
                fciDoc = fuFCIDoc.FileBytes;
            }

            DataRow dr = dt.NewRow();

            dr["GodownId"] = ddlGodown.SelectedValue;
            dr["GodownName"] = ddlGodown.SelectedItem.Text;

            dr["StackId"] = ddlStack.SelectedValue;
            dr["StackName"] = lblStackName.Text;

            dr["CommodityId"] = ViewState["CommodityId"].ToString();
            dr["Commodity"] = lblCommodity.Text;

            dr["Bags"] = lblBags.Text;
            dr["Qty"] = lblQty.Text;

            dr["MoistureDate"] = txtMoistureDate.Text.Trim();
            dr["AvgMoisture"] = txtAvgMoisture.Text.Trim();

            dr["SentFCI"] = ddlSentFCI.SelectedValue;
            dr["SentDate"] = txtSentDate.Text.Trim();

            dr["FCIChecked"] = ddlFCIChecked.SelectedValue;
            dr["FCICheckDate"] = txtFCICheckDate.Text.Trim();
            dr["FCIMoisture"] = txtFCIMoisture.Text.Trim();

            dr["BranchDocument"] = branchDoc;
            dr["FCIDocument"] = fciDoc;

            dt.Rows.Add(dr);

            Session["TempStack"] = dt;

            gvTempStack.DataSource = dt;
            gvTempStack.DataBind();

            ClearForNextStack();

            ShowMessage("Stack added successfully");
        }
        catch (Exception ex)
        {
            ShowMessage(ex.Message);
        }
    }
